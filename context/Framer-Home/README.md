For components with responsive variants, use the responsive runtime:

```tsx
// Import the CSS for responsive breakpoints
import './Button Icon/_responsive-runtime.css';

// Option 1: Use the useBreakpoint hook
import { useBreakpoint } from './Button Icon/_responsive-runtime';

function App() {
  const breakpoint = useBreakpoint(); // 'base' | 'sm' | 'md' | 'lg' | 'xl' | '2xl'

  return <ButtonIcon variant={breakpoint === 'base' ? 'mobile' : 'desktop'} />;
}

// Option 2: Use the WithBreakpoints HOC
import { WithBreakpoints } from './Button Icon/_responsive-runtime';

function App() {
  return (
    <WithBreakpoints
      Component={ButtonIcon}

        md: 'tablet',      // 768px+
        lg: 'desktop'      // 1024px+

    />
  );
}
```
